-- Extensions
CREATE EXTENSION IF NOT EXISTS "pgcrypto";  -- gen_random_uuid() dùng bên trong uuidv7()
CREATE EXTENSION IF NOT EXISTS "pg_trgm";   -- trigram full-text search

-- =============================================================
-- UUID v7 — Pure SQL, không cần extension
-- Tương thích Supabase (PG 17), không cần pg_uuidv7
-- Sortable by time, safe for distributed systems
-- =============================================================
CREATE OR REPLACE FUNCTION uuidv7()
RETURNS UUID AS $$
  SELECT encode(
    set_bit(
      set_bit(
        overlay(
          uuid_send(gen_random_uuid())
          placing substring(int8send(floor(extract(epoch FROM clock_timestamp()) * 1000)::BIGINT) FROM 3)
          FROM 1 FOR 6
        ),
        52, 1
      ),
      53, 1
    ),
    'hex'
  )::UUID;
$$ LANGUAGE SQL VOLATILE;

-- CUSTOMERS
-- Mỗi customer là 1 document, embed addresses & preferences
-- =============================================================
CREATE TABLE customers (
  id            UUID        PRIMARY KEY DEFAULT uuidv7(),
  email         TEXT        NOT NULL UNIQUE,
  phone         TEXT,

  -- Embedded document: thông tin cá nhân
  profile       JSONB       NOT NULL DEFAULT '{}',
  -- {
  --   "first_name": "Nguyen",
  --   "last_name": "Van A",
  --   "avatar_url": "https://...",
  --   "date_of_birth": "1995-03-15",
  --   "gender": "male"
  -- }

  -- Embedded array: nhiều địa chỉ, 1 cái là default
  addresses     JSONB       NOT NULL DEFAULT '[]',
  -- [
  --   {
  --     "id": "addr_001",
  --     "label": "Home",
  --     "is_default": true,
  --     "recipient": "Nguyen Van A",
  --     "phone": "0901234567",
  --     "street": "123 Nguyen Hue",
  --     "ward": "Ben Nghe",
  --     "district": "Quan 1",
  --     "city": "Ho Chi Minh",
  --     "country_code": "VN",
  --     "postal_code": "70000"
  --   }
  -- ]

  -- Embedded document: tuỳ chọn cá nhân
  preferences   JSONB       NOT NULL DEFAULT '{}',
  -- {
  --   "currency": "VND",
  --   "locale": "vi-VN",
  --   "newsletter": true,
  --   "sms_marketing": false,
  --   "tags": ["vip", "wholesale"]
  -- }

  -- Metadata: social logins, OAuth tokens, etc.
  metadata      JSONB       NOT NULL DEFAULT '{}',

  password_hash TEXT,
  is_active     BOOLEAN     NOT NULL DEFAULT TRUE,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at    TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Indexes
CREATE INDEX idx_customers_email      ON customers (email);
CREATE INDEX idx_customers_profile    ON customers USING GIN (profile);
CREATE INDEX idx_customers_addresses  ON customers USING GIN (addresses);
CREATE INDEX idx_customers_preferences ON customers USING GIN (preferences);
CREATE INDEX idx_customers_tags       ON customers USING GIN ((preferences -> 'tags'));

-- PRODUCTS

  -- [
  --   {
  --     "id": "var_001",
  --     "sku": "IP15P-256-BLK",
  --     "title": "256GB / Black Titanium",
  --     "price": 29990000,
  --     "compare_at_price": 32000000,
  --     "cost_price": 22000000,

  --     },
  --     "images": ["img_003"]
  --   }
  -- ]

  -- Embedded array: hình ảnh sản phẩm

  --   "og_image": "https://..."

  updated_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()

-- =============================================================
-- COLLECTIONS
-- Nhóm sản phẩm — manual (danh sách) hoặc smart (rule-based)

CREATE TABLE collections (
  id          UUID        PRIMARY KEY DEFAULT uuidv7(),
  slug        TEXT        NOT NULL UNIQUE,
  type        TEXT        NOT NULL DEFAULT 'manual'
                          CHECK (type IN ('manual', 'smart')),

  -- Embedded document: nội dung collection
  info        JSONB       NOT NULL DEFAULT '{}',
  -- {
  --   "title": "Flash Sale 6.6",
  --   "description": "Ưu đãi đặc biệt...",
  --   "image": { "url": "...", "alt": "..." },
  --   "published_at": "2025-06-06T00:00:00Z",
  --   "sort_order": "price_asc"
  -- }

  -- Smart collection: điều kiện lọc tự động
  rules       JSONB       NOT NULL DEFAULT '{}',
  -- {
  --   "match": "all",
  --   "conditions": [
  --     { "field": "tags",   "op": "contains", "value": "sale" },
  --     { "field": "price",  "op": "lt",        "value": 500000 },
  --     { "field": "status", "op": "eq",         "value": "active" }
  --   ]
  -- }

  seo         JSONB       NOT NULL DEFAULT '{}',

  -- Manual collection: danh sách product_id cụ thể (PostgreSQL native array)
  product_ids UUID[]      NOT NULL DEFAULT '{}',

  is_active   BOOLEAN     NOT NULL DEFAULT TRUE,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at  TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Indexes
CREATE INDEX idx_collections_slug        ON collections (slug);
CREATE INDEX idx_collections_type        ON collections (type);
CREATE INDEX idx_collections_product_ids ON collections USING GIN (product_ids);
CREATE INDEX idx_collections_rules       ON collections USING GIN (rules);


-- =============================================================
-- ORDERS
-- Toàn bộ line_items và địa chỉ EMBED vào order document
-- Snapshot customer tại thời điểm đặt hàng
-- =============================================================
CREATE TABLE orders (
  id              UUID        PRIMARY KEY DEFAULT uuidv7(),
  order_number    TEXT        NOT NULL UNIQUE,  -- "ORD-2025-00001"
  customer_id     UUID        REFERENCES customers(id) ON DELETE SET NULL,
  status          TEXT        NOT NULL DEFAULT 'pending'
                              CHECK (status IN (
                                'pending', 'confirmed', 'processing',
                                'shipped', 'delivered', 'cancelled', 'refunded'
                              )),

  -- Snapshot thông tin khách hàng tại lúc đặt hàng
  -- Không bị ảnh hưởng nếu customer update profile sau
  customer_snapshot JSONB     NOT NULL DEFAULT '{}',
  -- {
  --   "customer_id": "uuid...",
  --   "email": "customer@example.com",
  --   "phone": "0901234567",
  --   "name": "Nguyen Van A"
  -- }

  -- Embedded array: các sản phẩm trong đơn hàng
  line_items      JSONB       NOT NULL DEFAULT '[]',
  -- [
  --   {
  --     "line_id": "li_001",
  --     "product_id": "uuid...",
  --     "variant_id": "var_001",
  --     "sku": "IP15P-256-BLK",
  --     "title": "iPhone 15 Pro",
  --     "variant_title": "256GB / Black Titanium",
  --     "image_url": "https://...",
  --     "quantity": 2,
  --     "unit_price": 29990000,
  --     "discount_amount": 1000000,
  --     "final_price": 28990000,
  --     "currency": "VND",
  --     "fulfilled_qty": 0
  --   }
  -- ]

  -- Địa chỉ giao hàng — embedded doc
  shipping_address JSONB     NOT NULL DEFAULT '{}',
  -- {
  --   "recipient": "Nguyen Van A",
  --   "phone": "0901234567",
  --   "street": "123 Nguyen Hue",
  --   "ward": "Ben Nghe",
  --   "district": "Quan 1",
  --   "city": "Ho Chi Minh",
  --   "country_code": "VN"
  -- }

  -- Tóm tắt giá — embedded doc
  pricing         JSONB       NOT NULL DEFAULT '{}',
  -- {

-- =============================================================
-- PAYMENTS
-- Mỗi order có thể có nhiều payments (retry, split payment)
-- Gateway response raw data embed vào document
-- =============================================================
CREATE TABLE payments (
  id               UUID        PRIMARY KEY DEFAULT uuidv7(),
  order_id         UUID        NOT NULL REFERENCES orders(id) ON DELETE CASCADE,
  status           TEXT        NOT NULL DEFAULT 'pending'
                               CHECK (status IN (
                                 'pending', 'processing', 'paid',
                                 'failed', 'cancelled', 'refunded', 'partially_refunded'
                               )),

  -- Phương thức thanh toán — embedded doc
  method          JSONB        NOT NULL DEFAULT '{}',
  -- {
  --   "type": "card",          -- card | momo | zalopay | cod | bank_transfer | vnpay
  --   "provider": "stripe",
  --   "card_brand": "visa",
  --   "card_last4": "4242",
  --   "card_exp": "12/27",
  --   "wallet_phone": null,
  --   "bank_code": null
  -- }

  -- Số tiền — embedded doc (hỗ trợ multi-currency)
  amount          JSONB        NOT NULL DEFAULT '{}',
  -- {
  --   "value": 58010000,
  --   "currency": "VND",
  --   "exchange_rate": 1,
  --   "base_currency": "VND",
  --   "base_value": 58010000
  -- }

  -- Raw response từ payment gateway — giữ nguyên để audit
  gateway_response JSONB       NOT NULL DEFAULT '{}',
  -- {
  --   "gateway": "vnpay",
  --   "transaction_id": "VNP_TXN_001",
  --   "response_code": "00",
  --   "message": "Success",
  --   "bank_code": "VCB",
  --   "pay_date": "20250606120000",
  --   "raw": { ...full gateway payload... }
  -- }

  -- Lịch sử hoàn tiền — embedded array
  refunds         JSONB        NOT NULL DEFAULT '[]',
  -- [
  --   {
  --     "refund_id": "ref_001",
  --     "amount": 29990000,
  --     "currency": "VND",
  --     "reason": "Customer cancelled",
  --     "status": "completed",
  --     "gateway_refund_id": "re_xxx",
  --     "created_at": "2025-06-08T10:00:00Z"
  --   }
  -- ]

  metadata        JSONB        NOT NULL DEFAULT '{}',

  paid_at         TIMESTAMPTZ,
  created_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
  updated_at      TIMESTAMPTZ  NOT NULL DEFAULT NOW()
);

-- Indexes
CREATE INDEX idx_payments_order_id  ON payments (order_id);
CREATE INDEX idx_payments_status    ON payments (status);
CREATE INDEX idx_payments_paid_at   ON payments (paid_at DESC);
CREATE INDEX idx_payments_method    ON payments USING GIN (method);
CREATE INDEX idx_payments_gateway   ON payments USING GIN (gateway_response);
CREATE INDEX idx_payments_refunds   ON payments USING GIN (refunds);


-- =============================================================
-- UTILITY: Auto-update updated_at trigger
-- =============================================================
CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_customers_updated_at
  BEFORE UPDATE ON customers
  FOR EACH ROW EXECUTE FUNCTION set_updated_at();

CREATE TRIGGER trg_products_updated_at
  BEFORE UPDATE ON products
  FOR EACH ROW EXECUTE FUNCTION set_updated_at();

CREATE TRIGGER trg_collections_updated_at
  BEFORE UPDATE ON collections
  FOR EACH ROW EXECUTE FUNCTION set_updated_at();

CREATE TRIGGER trg_orders_updated_at
  BEFORE UPDATE ON orders
  FOR EACH ROW EXECUTE FUNCTION set_updated_at();

CREATE TRIGGER trg_payments_updated_at
  BEFORE UPDATE ON payments
  FOR EACH ROW EXECUTE FUNCTION set_updated_at();
